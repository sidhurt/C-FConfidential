FORM authority_check .
  IF r_werks[] IS NOT INITIAL.
    DATA werks  TYPE RANGE OF werks WITH HEADER LINE.
    REFRESH werks.
    DATA : lv_message  TYPE string.
    CLEAR lv_message.
    SELECT FROM t001w
      FIELDS DISTINCT
      werks
      WHERE werks IN @r_werks
      INTO   TABLE @DATA(it_plant).
    IF sy-subrc NE 0.
      MESSAGE 'Plant Not Found' TYPE 'E'.
    ENDIF.

    LOOP AT it_plant INTO DATA(wa).
      AUTHORITY-CHECK OBJECT 'ZLE_PLNT'
      ID 'WERKS' FIELD wa-werks.
      IF sy-subrc <> 0.
        r_werks-low = wa-werks.
        r_werks-high = ''.
        r_werks-option = 'BT'.
        r_werks-sign = 'I'.
        APPEND werks.
        CONCATENATE wa-werks ','  lv_message INTO lv_message SEPARATED BY space.
        CONDENSE  lv_message NO-GAPS.
*      DELETE s_werks WHERE low = wa-werks.
        CLEAR wa.
      ENDIF.
    ENDLOOP.
    IF werks[] IS NOT INITIAL."""s_werks[] IS INITIAL OR
      REFRESH r_werks[].
      DATA(lv_message1) = CONV string( 'You Are Not Authorised For' ).
      DATA(lv_message2) = CONV string( 'Plant' ).
      CONCATENATE lv_message1 lv_message lv_message2 INTO DATA(message) SEPARATED BY space.
*    DATA(flag) = 'X'.
      MESSAGE  message  TYPE 'E' DISPLAY LIKE 'I'.
    ENDIF.
  ENDIF.

ENDFORM.

                  spart FOR vbak-spart NO-EXTENSION NO INTERVALS OBLIGATORY,
                  vkbur FOR vbak-vkbur,
                  werks FOR vbap-werks,
                  matnr FOR vbap-matnr,
                  land1 FOR kna1-land1 DEFAULT 'IN' NO-DISPLAY,
                  regio FOR kna1-regio,
                  city2 FOR zsdtprice-zdist_ext.
  PARAMETERS : chk  AS CHECKBOX,
               chk1 AS CHECKBOX,
               chk2 AS CHECKBOX.
SELECTION-SCREEN END OF BLOCK b1.
SELECTION-SCREEN:BEGIN OF BLOCK b2 WITH FRAME TITLE TEXT-ss2.
  PARAMETERS:p_layout TYPE slis_vari.
SELECTION-SCREEN:END OF BLOCK b2.

  PERFORM find_layout.

  CLEAR: gv_receive_jobs,

  CALL FUNCTION 'FIMA_DAYS_AND_MONTHS_AND_YEARS'

    IMPORTING

FORM f_init_server_group.
  CALL FUNCTION 'SPBT_INITIALIZE'
    EXPORTING
      group_name                     = gc_rfcgr
*     max_wait_time                  = 0
    IMPORTING
      max_pbt_wps                    = gv_total
      free_pbt_wps                   = gv_available
    EXCEPTIONS
      invalid_group_name             = 1
      internal_error                 = 2
      pbt_env_already_initialized    = 3
      currently_no_resources_avail   = 4
      no_pbt_resources_found         = 5
      cant_init_different_pbt_groups = 6
      OTHERS                         = 7.
  CASE sy-subrc.
    WHEN 4.
      gv_initialisation_ok = ' '.
    WHEN 0 OR 3.
      max_use = SWITCH #( sy-sysid WHEN 'DS4' THEN 6
                                   WHEN 'QS4' THEN 15
                                   WHEN 'PS4' THEN 15 ).
      gv_initialisation_ok = 'X'.
      gv_available = COND #( WHEN gv_available GE max_use THEN max_use ELSE gv_available ).
    WHEN 1 OR 2 OR 5.
      WRITE:/ |Server group { gc_rfcgr } could not be initialized|.
      EXIT.
  ENDCASE.
ENDFORM.
FORM f_parallel_get_details_mif_in.
  cl_progress_indicator=>progress_indicate( i_text = 'Processing Data' ).
  RANGES: r_erdat FOR sy-datum.
  LOOP AT erdat WHERE high IS INITIAL.
    erdat-high = erdat-low.
    erdat-option = 'BT'.
    MODIFY erdat.
  ENDLOOP.
  DO.
    IF gv_initialisation_ok EQ abap_true.
      CALL FUNCTION 'SPBT_GET_CURR_RESOURCE_INFO'
        IMPORTING
          max_pbt_wps                 = lv_max_wps
          free_pbt_wps                = lv_free_wps
        EXCEPTIONS
          internal_error              = 1
          pbt_env_not_initialized_yet = 2
          OTHERS                      = 3.
      IF lv_free_wps GT 2 AND gv_pp_running LT lv_wproc.
      ELSE.
        WAIT UP TO 1 SECONDS.
        CONTINUE.
      ENDIF.
      IF r_erdat[] IS INITIAL.
        r_erdat = VALUE #( sign = 'I' option = 'EQ' low = erdat-low high = erdat-high ).
        APPEND r_erdat.
      ELSE.
        CALL FUNCTION 'FIAPPL_ADD_DAYS_TO_DATE'
          EXPORTING
            i_date      = r_erdat-low
            i_days      = 1
          IMPORTING
            e_calc_date = r_erdat-low.
        r_erdat-sign = 'I'.
        r_erdat-option = 'BT'.
        r_erdat-high = r_erdat-high.
        FREE: r_erdat[].
        APPEND r_erdat.
      ENDIF.
      IF r_erdat-low LE erdat-high.

        gv_taskname = gv_taskname + 1.
        gv_pp_running = gv_pp_running + 1.

        gv_result = 3.
        WHILE gv_result NE 0.
          lv_from += 1.
          cl_progress_indicator=>progress_indicate( i_text = 'Processing Data' i_processed = lv_from i_total = lv_to ).

          CALL FUNCTION 'ZSD_PENDING_ORDER_FM'
            STARTING NEW TASK gv_taskname
            DESTINATION IN GROUP gc_rfcgr
            PERFORMING f_call_back_mif_in ON END OF TASK
            EXPORTING
              vbeln = vbeln[]
              auart = auart[]
              erdat = r_erdat[]
              kunnr = kunnr[]
              vkorg = vkorg[]
              vtweg = vtweg[]
              spart = spart[]
              vkbur = vkbur[]
              werks = werks[]
              matnr = matnr[]
              regio = regio[]
              city2 = city2[]
              chk   = chk
              chk1  = chk1
              chk2  = chk2.

          gv_result = sy-subrc.
          CASE gv_result.
            WHEN 0.
              gv_send_jobs += 1.
              gv_jobs = gv_send_jobs - gv_receive_jobs.
              CONTINUE.
            WHEN OTHERS.
              WAIT UP TO 1 SECONDS.
          ENDCASE.
        ENDWHILE.
      ENDIF.
    ELSE.
      PERFORM f_init_server_group.
    ENDIF.
    IF r_erdat[ 1 ]-low GT r_erdat[ 1 ]-high.
      EXIT.
    ENDIF.
  ENDDO.
  WAIT UNTIL gv_receive_jobs >= gv_send_jobs UP TO 1200 SECONDS.
ENDFORM.
FORM find_layout .
  DATA(ls_alv_variant) = VALUE disvariant( ).
  ls_alv_variant-report = sy-repid.
  CALL FUNCTION 'REUSE_ALV_VARIANT_F4'
    EXPORTING
      is_variant = ls_alv_variant
      i_save     = 'A'
    IMPORTING
      es_variant = ls_alv_variant
    EXCEPTIONS
      not_found  = 2.
  IF sy-subrc = 2.
    MESSAGE ID sy-msgid TYPE 'S' NUMBER sy-msgno
    WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
  ELSE.
    p_layout = ls_alv_variant-variant.
  ENDIF.
ENDFORM.
FORM f_call_back_mif_in USING gv_taskname.
  DATA: it_table LIKE lt_final[].

  gv_receive_jobs += 1.
  gv_jobs = gv_send_jobs - gv_receive_jobs.

  RECEIVE RESULTS FROM FUNCTION 'ZSD_PENDING_ORDER_FM'
  TABLES lt_final = it_table.
  APPEND LINES OF it_table TO lt_final.
  FREE: it_table.

  gv_pp_running -= 1.

ENDFORM.
FORM process_data.
  DATA: it_data LIKE lt_final[].
  PERFORM f_parallel_get_details_mif_in.
  IF lt_final[] IS NOT INITIAL.
    PERFORM display USING lt_final[].
  ELSE.
    MESSAGE 'Data Not Found' TYPE 'E'.
  ENDIF.
ENDFORM.
FORM auth_check.

ENDFORM.
