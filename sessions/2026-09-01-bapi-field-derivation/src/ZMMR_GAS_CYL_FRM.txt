*&---------------------------------------------------------------------*
*& Form get_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
  form get_data .
    select * from objk
      into table @data(lt_objk)
     where matnr = @p_mat
       and sernr in @s_ser .

    sort lt_objk by equnr sernr matnr.
    delete adjacent duplicates from lt_objk comparing equnr sernr matnr .

    if lt_objk is not initial .
      select * from eqbs into table @data(lt_eqbs)
      for all entries in @lt_objk
      where equnr = @lt_objk-equnr
      and b_werk = @p_werks .
      if lt_eqbs is not initial.
        select * from equi into table @data(lt_equi)
        for all entries in @lt_eqbs
        where equnr = @lt_eqbs-equnr .
        if lt_equi is not initial.
          select * from jest into table @data(lt_jest)
          for all entries in @lt_equi
          where objnr = @lt_equi-objnr
            and inact = ' '
            and stat like 'E%'.

          sort lt_jest by objnr chgnr descending .

          if lt_jest is not initial .
            select * from tj30t into table @data(lt_tj30t)
              for all entries in @lt_jest
              where estat = @lt_jest-stat
              and stsma = 'PM_EQUP1'
              and spras = @sy-langu .

          endif.
        endif.
      endif.
    endif.

    loop at lt_objk into data(lw_objk).
      read table lt_eqbs into data(lw_eqbs) with key equnr = lw_objk-equnr .
      if sy-subrc = 0.
        gw_final-matnr = lw_objk-matnr .
        shift gw_final-matnr left deleting leading '0'.
        gw_final-sernr = lw_objk-sernr .
        shift gw_final-sernr left deleting leading '0'.
        gw_final-equnr = lw_objk-equnr .
        shift gw_final-equnr left deleting leading '0'.
        gw_final-werks = lw_eqbs-b_werk .
        gw_final-sloc = lw_eqbs-b_lager .
        read table lt_equi into data(lw_equi) with key equnr = lw_objk-equnr.
        if sy-subrc = 0.
          gw_final-objnr = lw_equi-objnr .
          read table lt_jest into data(lw_jest) with key objnr = lw_equi-objnr .
          if sy-subrc = 0.
            gw_final-stat = lw_jest-stat .
            read table lt_tj30t into data(lw_tj30t) with key estat = lw_jest-stat .
            if sy-subrc = 0.
              gw_final-cstat = lw_tj30t-txt04.
            endif.
          endif.
        endif.
      endif.
      append gw_final to gt_final.
      clear gw_final.
    endloop.

    delete gt_final where matnr = ' '.
  endform.
