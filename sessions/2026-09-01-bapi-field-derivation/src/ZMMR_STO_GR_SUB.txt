    EXPORTING
      i_callback_program       = lv_repid
      is_layout                = gs_layout
      it_fieldcat              = gt_fieldcat

    EXCEPTIONS
      program_error            = 1
      OTHERS                   = 2.

    MESSAGE c_msg1 TYPE c_e.
  ENDIF.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form f_user_command
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM f_user_command USING p_ucomm LIKE sy-ucomm
                           p_selfield TYPE slis_selfield.
  DATA ref TYPE REF TO cl_gui_alv_grid.
  CALL FUNCTION 'GET_GLOBALS_FROM_SLVC_FULLSCR'
    IMPORTING
      e_grid = ref.
  CALL METHOD ref->check_changed_data.

  CASE p_ucomm.
    WHEN c_post.
      PERFORM f_post_data.
    WHEN c_sela.
      LOOP AT gt_final INTO gs_final.
        gs_final-sel = abap_true.
        MODIFY gt_final FROM gs_final.
      ENDLOOP.
    WHEN c_dsela.
      LOOP AT gt_final INTO gs_final.
        gs_final-sel = abap_false.
        MODIFY gt_final FROM gs_final.
      ENDLOOP.
    WHEN OTHERS.
  ENDCASE.
  p_selfield-refresh = abap_true.
ENDFORM.

*&---------------------------------------------------------------------*
*& Form f_post_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM f_post_data.
  TYPES : BEGIN OF ty_s_gr,
            vbeln TYPE vbeln_vf,
            mblnr TYPE mblnr,
            msg   TYPE char200,
          END OF ty_s_gr.

  DATA : ls_goodsmvt_header  TYPE bapi2017_gm_head_01,
         ls_goodsmvt_code    TYPE bapi2017_gm_code,
         ls_goodsmvt_headret TYPE bapi2017_gm_head_ret,
         lv_materialdocument TYPE bapi2017_gm_head_ret-mat_doc,
         lv_matdocumentyear  TYPE bapi2017_gm_head_ret-doc_year,
         lt_goodsmvt_item    TYPE TABLE OF bapi2017_gm_item_create,
         ls_goodsmvt_item    TYPE bapi2017_gm_item_create,
         lt_return           TYPE TABLE OF bapiret2,
         ls_return           TYPE bapiret2,
         lt_gr               TYPE STANDARD TABLE OF ty_s_gr,
         ls_gr               TYPE ty_s_gr,
         line_id             TYPE i.

  CLEAR lt_gr[].
  LOOP AT gt_final INTO DATA(gs_final1) WHERE sel = abap_true AND mblnr IS INITIAL.
    CLEAR gs_final.
    gs_final = gs_final1.
    AT NEW vbeln.
      CLEAR : ls_goodsmvt_code,ls_goodsmvt_header,line_id.
      ls_goodsmvt_code              = '01'.
      ls_goodsmvt_header-pstng_date = gs_final-wadat_ist.
      ls_goodsmvt_header-doc_date   = gs_final-wadat_ist.
      ls_goodsmvt_header-ref_doc_no = gs_final-vbeln.
    ENDAT.

    CLEAR : ls_goodsmvt_item,lt_goodsmvt_item[].
    line_id = line_id + 1.
    ls_goodsmvt_item-line_id    = line_id.
    ls_goodsmvt_item-material   = gs_final-matnr.
    ls_goodsmvt_item-plant      = gs_final-werks.
    ls_goodsmvt_item-stge_loc   = gs_final-lgort.
    ls_goodsmvt_item-batch      = gs_final-charg.
    ls_goodsmvt_item-move_type  = '101'.
    ls_goodsmvt_item-entry_qnt  = gs_final-lfimg.
    ls_goodsmvt_item-quantity   = gs_final-lfimg.
    ls_goodsmvt_item-entry_uom  = gs_final-vrkme.
    ls_goodsmvt_item-base_uom   = gs_final-meins.
    ls_goodsmvt_item-po_number  = gs_final-ebeln.
    ls_goodsmvt_item-po_item    = gs_final-ebelp.
    ls_goodsmvt_item-deliv_numb = gs_final-vbeln.
    ls_goodsmvt_item-deliv_item = gs_final-posnr.
    ls_goodsmvt_item-mvt_ind    = 'B'.
    APPEND ls_goodsmvt_item TO lt_goodsmvt_item.

    AT END OF vbeln.
      CHECK lt_goodsmvt_item[] IS NOT INITIAL.
      CLEAR: lt_return.
      CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
        EXPORTING
          goodsmvt_header  = ls_goodsmvt_header
          goodsmvt_code    = ls_goodsmvt_code
*         testrun          = 'X'
        IMPORTING
          goodsmvt_headret = ls_goodsmvt_headret
          materialdocument = lv_materialdocument
          matdocumentyear  = lv_matdocumentyear
        TABLES
          goodsmvt_item    = lt_goodsmvt_item[]
          return           = lt_return[].

      IF lv_materialdocument IS NOT INITIAL .
        gs_final-mblnr =  lv_materialdocument.
        CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
          EXPORTING
            wait = 'X'.
        WAIT UP TO 2 SECONDS .
        gs_final-msg = 'GR Created'.

        CLEAR ls_gr.
        ls_gr-vbeln = gs_final-vbeln.
        ls_gr-mblnr = gs_final-mblnr.
        ls_gr-msg   = 'GR Created'.
        APPEND ls_gr TO lt_gr.
      ELSE.
        CLEAR ls_return.
        READ TABLE lt_return INTO ls_return WITH KEY type = c_e.
        IF sy-subrc = 0.
          CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
          gs_final-msg = ls_return-message.

          CLEAR ls_gr.
          ls_gr-vbeln = gs_final-vbeln.
          ls_gr-msg   = gs_final-msg.
          APPEND ls_gr TO lt_gr.
        ENDIF.
      ENDIF.
      MODIFY gt_final FROM gs_final.
    ENDAT.
  ENDLOOP.
  LOOP AT gt_final INTO gs_final WHERE sel = abap_true.
    CLEAR ls_gr.
    READ TABLE lt_gr INTO ls_gr WITH KEY vbeln = gs_final-vbeln.
    IF sy-subrc = 0.
      gs_final-mblnr  = ls_gr-mblnr.
      gs_final-msg    = ls_gr-msg.
      MODIFY gt_final FROM gs_final TRANSPORTING mblnr msg.
    ENDIF.
  ENDLOOP.
ENDFORM.
