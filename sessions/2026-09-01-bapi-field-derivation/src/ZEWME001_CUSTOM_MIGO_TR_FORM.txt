FORM Readdata .
  DATA:
*        it_matnr TYPE  rsdsselopt_t,
*        wa_matnr LIKE LINE OF it_matnr,
        lchk     TYPE i.
  SELECT   a~matnr,
           a~charg,
           a~menge,
           a~meins,
           a~werks,
           a~lgort,
    CASE WHEN b~bednr = 'POWER' THEN  'WH01'
         WHEN b~bednr = 'CEMENT' THEN  'WH01'
          END AS umlgo,
         '311' AS  bwart,
    a~zeile AS zeile,
    a~bwtar AS bwtar
  FROM mseg AS a
LEFT OUTER JOIN ekpo AS a1
ON a~ebeln = a1~ebeln  AND
   a~ebelp = a1~ebelp
LEFT OUTER JOIN eban AS b
ON a1~banfn = b~banfn AND
   a1~bnfpo = b~bnfpo AND
    a1~bednr = b~bednr
INTO CORRESPONDING FIELDS OF TABLE @it_final
  WHERE a~werks = @p_werks AND
        a~mblnr = @p_mblnr AND
        a~mjahr = @p_mjahr AND
        a~budat_mkpf IN @s_budat.

  IF it_final[] IS NOT INITIAL.
    SELECT plant ,
      stge_loc,
      lgnum  FROM /scwm/tmapstloc
      INTO TABLE @DATA(it_data)
      FOR ALL ENTRIES IN @it_final
      WHERE plant = @it_final-werks .

    SELECT matnr,werks,sernp FROM marc
    INTO TABLE @DATA(it_serial)
    FOR ALL ENTRIES IN @it_final
    WHERE matnr =  @it_final-matnr AND
    werks = @it_final-werks.

    LOOP AT it_final ASSIGNING FIELD-SYMBOL(<lfs_final>).
      <lfs_final>-mblnr = p_mblnr.
      <lfs_final>-lgnum = VALUE #( it_data[ plant = <lfs_final>-werks ]-lgnum OPTIONAL ).
      <lfs_final>-sernp = VALUE #( it_serial[ matnr = <lfs_final>-matnr
      werks = <lfs_final>-werks ]-sernp OPTIONAL ).
    ENDLOOP.


    SELECT lgnum,lgpla,matnr FROM /scwm/binmat
      INTO TABLE @gt_sto
     FOR ALL ENTRIES IN @it_final
    WHERE
       matnr =  @it_final-matnr AND
      lgnum = @it_final-lgnum.

    DELETE ADJACENT DUPLICATES FROM gt_sto.
    LOOP AT it_final ASSIGNING <lfs_final>.
      CLEAR lchk.
      LOOP AT gt_sto INTO gs_sto WHERE matnr = <lfs_final>-matnr AND
       lgnum = <lfs_final>-lgnum .
        lchk = lchk + 1.
      ENDLOOP.
      IF lchk = 1.
        <lfs_final>-lgpla =  gs_sto-lgpla.
      ENDIF.
    ENDLOOP.

  ENDIF.
*         lgpla TYPE /scwm/lgpla,
ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM F4_documentnum .
  DATA: it_mblnr TYPE RANGE OF mblnr,
        lday(2),
        lmon(2),
        lyear(4),
        wa_mblnr LIKE LINE OF it_mblnr.
  DATA: lt_ret        TYPE STANDARD TABLE OF ddshretval,
        s_budat1      TYPE RANGE OF budat WITH HEADER LINE,
        dynpro_values TYPE TABLE OF dynpread,
        field_value   LIKE LINE OF dynpro_values.
  CLEAR: field_value, dynpro_values. REFRESH dynpro_values.
  field_value-fieldname = 'P_WERKS'.
  APPEND field_value TO dynpro_values.
  field_value-fieldname = 'S_BUDAT-LOW'.
  APPEND field_value TO dynpro_values.
  field_value-fieldname = 'S_BUDAT-HIGH'.
  APPEND field_value TO dynpro_values.
  CALL FUNCTION 'DYNP_VALUES_READ'
    EXPORTING
      dyname             = sy-repid
      dynumb             = sy-dynnr
      translate_to_upper = 'X'
    TABLES
      dynpfields         = dynpro_values.

  READ TABLE dynpro_values INDEX 1 INTO field_value.
  IF sy-subrc = 0.
    p_werks =  field_value-fieldvalue.
  ENDIF.
  CLEAR: lday ,lmon ,lyear.
  READ TABLE dynpro_values INDEX 2 INTO field_value.
  IF sy-subrc = 0.
    SPLIT field_value-fieldvalue AT '.' INTO
    lday lmon lyear.
    CONCATENATE lyear lmon lday INTO s_budat1-low.
  ENDIF.
  READ TABLE dynpro_values INDEX 3 INTO field_value.
  IF sy-subrc = 0.
    SPLIT field_value-fieldvalue AT '.' INTO
    lday lmon lyear.
    CONCATENATE lyear lmon lday INTO s_budat1-high.
  ENDIF.
  s_budat1-sign = 'I'.
  IF s_budat1-low IS NOT INITIAL AND s_budat1-high IS NOT INITIAL.
    s_budat1-option =  'BT'.
  ELSE.
    s_budat1-option =  'EQ'.
  ENDIF.
  APPEND s_budat1.
  SELECT DISTINCT mblnr FROM matdoc
  INTO TABLE @DATA(it_matdoc)
  WHERE werks = @p_werks AND
  budat IN @s_budat1 AND
    ( bwart  = '101' OR  bwart  = '105' ).

  wa_mblnr-sign =  'I'.
  wa_mblnr-option =  'EQ'.

  LOOP AT it_matdoc INTO DATA(wa_matdoc).
    wa_mblnr-low = wa_matdoc-mblnr.
    APPEND wa_mblnr TO it_mblnr.
  ENDLOOP.

  SELECT DISTINCT mblnr,bktxt FROM matdoc
  INTO TABLE @DATA(lt_matdoc)
  WHERE bktxt IN @it_mblnr.

  LOOP AT lt_matdoc INTO DATA(ls_matdoc).
    DELETE it_matdoc WHERE mblnr = ls_matdoc-bktxt.
  ENDLOOP.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield        = 'MBLNR'
      dynpprog        = sy-repid
      dynpnr          = sy-dynnr
      dynprofield     = 'P_MBLNR'
      value_org       = 'S'
    TABLES
      value_tab       = it_matdoc[] "F4 value list table
      return_tab      = lt_ret[]  "user selected table
    EXCEPTIONS
      parameter_error = 1
      no_values_found = 2
      OTHERS          = 3.

  IF sy-subrc = 0.
    LOOP AT lt_ret INTO DATA(ls_sel).
      p_mblnr     = ls_sel-fieldval.
    ENDLOOP.
  ENDIF.
ENDFORM.
