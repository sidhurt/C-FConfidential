  CLEAR gt_exl[].
  CALL FUNCTION 'ALSM_EXCEL_TO_INTERNAL_TABLE'
    EXPORTING
      filename                = p_file
      i_begin_col             = 1
      i_begin_row             = 1                "2-1   "p_begin
      i_end_col               = 24
      i_end_row               = 50000       "p_end
    TABLES
      intern                  = gt_exl
    EXCEPTIONS
      inconsistent_parameters = 1
      upload_ole              = 2
      OTHERS                  = 3.
  DATA: lv_sno TYPE i.
  CLEAR lv_sno.
  CLEAR: lv_xlcols.
  LOOP AT gt_exl INTO DATA(gs_exlcol) WHERE row = 1.
    lv_xlcols = lv_xlcols + 1.
  ENDLOOP.
  DATA:n            TYPE i.
*  DESCRIBE TABLE gt_ LINES n.
*  IF  lv_xlcols = n.
  IF gt_exl[] IS NOT INITIAL.
    LOOP AT gt_exl INTO DATA(lw_exl) WHERE row > 1.
      CASE lw_exl-col.
        WHEN '0001'.
          gs_file-doc_type = lw_exl-value.
        WHEN '0002'.
          gs_file-ref_doc  = lw_exl-value.
        WHEN '0003'.
          gs_file-ebeln = lw_exl-value.
        WHEN '0004'.
          gs_file-ebelp = lw_exl-value.

*          gs_file-bldat = lw_exl-value.
        WHEN '0006'.
          CONCATENATE lw_exl-value+6(4) lw_exl-value+3(2) lw_exl-value+0(2) INTO gs_file-budat .
*          gs_file-bldat = lw_exl-value.
*          gs_file-budat = lw_exl-value.
        WHEN '0007'.
          gs_file-xblnr  = lw_exl-value.
        WHEN '0008'.
          gs_file-frbnr  = lw_exl-value.
        WHEN '0009'.
          gs_file-bktxt = lw_exl-value.
        WHEN '0010'.
          gs_file-token = lw_exl-value.
        WHEN '0011'.
          gs_file-lrdat = lw_exl-value.
        WHEN '0012'.
          gs_file-afrno = lw_exl-value.
        WHEN '0013'.
          gs_file-ebill = lw_exl-value.
        WHEN '0014'.
          gs_file-afrdt = lw_exl-value.
        WHEN '0015'.
          gs_file-edate = lw_exl-value.
        WHEN '0016'.
          gs_file-lovct = lw_exl-value.
        WHEN '0017'.
          gs_file-bwart = lw_exl-value.
        WHEN '0018'.
          gs_file-menge = lw_exl-value.
        WHEN '0019'.
          gs_file-meins = lw_exl-value.
        WHEN '0020'.
          gs_file-lgort = lw_exl-value.
        WHEN '0021'.
          gs_file-charg = lw_exl-value.
*        WHEN '0022'.
*          gs_file-item_ok = lw_exl-value.
*        WHEN '0023'.
*          gs_file-check = lw_exl-value.
*        WHEN '0024'.
*          gs_file-save = lw_exl-value.
      ENDCASE.
      AT END OF row.
        APPEND gs_file TO gt_file.
        CLEAR:gs_file.
      ENDAT.
    ENDLOOP.
  ENDIF.
  CLEAR gs_file.
*  BREAK ibmabap32.
  LOOP AT gt_file INTO gs_file.
*    *****header************
    CLEAR: ls_bapi_header.
    IF gs_file-budat IS NOT INITIAL AND gs_file-bldat IS NOT INITIAL.
      ls_bapi_header-pstng_date = gs_file-budat.
      ls_bapi_header-doc_date = gs_file-bldat.

*    ls_bapi_header-header_txt  = gs_file-bktxt.

***     item tab**********
    CLEAR:lt_bapi_item[].
    ls_bapi_item-mvt_ind = 'B'.
*    ls_bapi_item-deliv_numb = gs_file-xblnr.

*    ls_bapi_item-
    APPEND ls_bapi_item TO lt_bapi_item.
*    ****************
    CLEAR:lt_return[],lv_document,lv_document.
    CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
      EXPORTING
        goodsmvt_header  = ls_bapi_header
        goodsmvt_code    = '01'
      IMPORTING
        materialdocument = lv_document
        matdocumentyear  = lv_documentyear
      TABLES
        goodsmvt_item    = lt_bapi_item
        extensionin      = lt_bapi_extension_in
        return           = lt_return.

    IF lv_document IS NOT INITIAL.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          wait = abap_true.
      MOVE-CORRESPONDING gs_file TO gs_final.
      CONCATENATE 'Successful:'lv_document 'is posted' INTO gs_final-status SEPARATED BY space.
      gs_final-year = lv_documentyear.
      gs_final-mat_doc = lv_document.
      APPEND gs_final TO gt_final.
    ELSE.
      MOVE-CORRESPONDING gs_file TO gs_final.
      READ TABLE lt_return INTO DATA(ls_return) WITH KEY  type  = 'E'.
      IF sy-subrc EQ 0.
        CONCATENATE 'Unsuccessful:'ls_return-message INTO gs_final-status SEPARATED BY space.
      ENDIF.
      APPEND gs_final TO gt_final.
      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    ENDIF.
  ENDLOOP.
ENDFORM.

      .

      ENDIF.
    ELSE.
