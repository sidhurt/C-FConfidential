  METHOD post_cancl_cust.
    SELECT credit_note, debit_note, db_bill_doc, cr_bill_doc
      INTO TABLE @DATA(lt_lookup)
      FROM zsd_bag_dr_cr
      FOR ALL ENTRIES IN @it_final_temp
    WHERE vbeln EQ @it_final_temp-vbeln
      AND vgbel EQ @it_final_temp-vgbel
      AND canc_flg EQ 'X'
      AND new_cr_dr EQ ' '.
    IF sy-subrc IS INITIAL.
      DATA(lt_lookup_cr) = lt_lookup.
      SORT lt_lookup_cr BY credit_note .
      SORT lt_lookup BY debit_note.
    ENDIF.
    CLEAR lv_int.
    LOOP AT it_final_temp INTO ls_final.
      CLEAR: lv_int, lv_srno.
      lv_srno = ls_final-s_no.
      MOVE-CORRESPONDING ls_final TO ls_final_temp1.
      CLEAR ls_datain.
      lv_int += 10.
      ls_datain-salesorg    = ls_final_temp1-vkorg.
      ls_datain-distr_chan  = ls_final_temp1-vtweg.
      ls_datain-division    = ls_final_temp1-spart.
      ls_datain-plant       = ls_final_temp1-werks.
      ls_datain-bill_date   = sy-datum.
      ls_datain-price_date  = sy-datum.
      ls_datain-material    = ls_final_temp1-matnr.
      ls_datain-itm_number  = ls_final_temp1-item.
      ls_datain-ref_item    = lv_int.
      ls_datain-ref_doc     = ls_final_temp1-sale_order.
      ls_datain-doc_number  = ls_final_temp1-sale_order.
      ls_datain-ref_doc_ca  = 'K'.

      CLEAR ls_update.
      ls_update-vbeln = ls_final_temp1-vbeln.
      ls_update-vgbel = ls_final_temp1-vgbel.
      ls_update-inv_qty = ls_final_temp1-inv_qty.
      ls_update-act_gross_weight = ls_final_temp1-act_gross_weight.
      ls_update-act_tare_weight = ls_final_temp1-act_tare_weight.
      ls_update-act_net_weight = ls_final_temp1-act_net_weight.
      ls_update-short_qty_bag = ls_final_temp1-short_qty_bag.
      ls_update-short_qty_mt = ls_final_temp1-short_qty_mt.
      ls_update-short_tol_per = ls_final_temp1-short_tol_per.
      ls_update-damage_qty_bag = ls_final_temp1-damage_qty_bag.
      ls_update-damage_qty_mt = ls_final_temp1-damage_qty_mt.
      ls_update-damage_tol_mt = ls_final_temp1-damage_tol_mt.
      ls_update-damage_tol_per = ls_final_temp1-damage_tol_per.
      ls_update-short_net_weight = ls_final_temp1-short_net_weight.
      ls_update-drn_rate = ls_final_temp1-drn_rate.
      ls_update-amt_drn_to_tranp = ls_final_temp1-amt_drn_to_tranp.
      ls_update-remarks = ls_final_temp1-remarks.
      ls_update-created_by = sy-uname.
      ls_update-created_on = sy-datum.
      ls_update-created_time = sy-uzeit.
      ls_update-s_no = lv_srno.
      ls_update-crn_rate = ls_final_temp1-crn_rate.
      CASE ls_final-cr_dr_ind.
        WHEN 'CREDIT'.
          ls_datain-doc_type   = 'ZSCR'.
          ls_datain-sold_to    = |{ ls_final_temp1-kunnr ALPHA = IN }|.
          ls_datain-ship_to    = |{ ls_final_temp1-kunnr ALPHA = IN }|.
          APPEND ls_datain TO lt_datain.
          CLEAR: ls_datain.
          REFRESH: lt_return1, lt_success.
          CALL FUNCTION 'BAPI_BILLINGDOC_CREATEMULTIPLE'
            EXPORTING
              creatordatain = ls_create
              testrun       = ' '
            TABLES
              billingdatain = lt_datain
              return        = lt_return1
              success       = lt_success.
          SORT lt_success BY ref_doc.
          READ TABLE lt_success INTO DATA(ls_success) WITH KEY ref_doc = ls_final_temp1-sale_order BINARY SEARCH.
          IF sy-subrc = 0.
            DATA(lv_bill1)      = ls_success-bill_doc.
            ls_final_temp1-crn_doc_date = sy-datum.
            ls_final_temp1-crn_doc_no = lv_bill1.
            ls_final_temp1-credit_qty = ls_final_temp1-short_net_weight.

            ls_update-credit_note = ls_final_temp1-sale_order.

            READ TABLE lt_lookup_cr INTO DATA(ls_lookup) WITH KEY credit_note = ls_update-credit_note BINARY SEARCH.
            IF sy-subrc IS INITIAL.
              UPDATE zsd_bag_dr_cr
              SET
               new_cr_dr = lv_bill1
              WHERE vbeln = ls_final_temp1-vbeln
               AND vgbel = ls_final_temp1-vgbel
               AND credit_note = ls_final_temp1-sale_order
               AND cr_bill_doc = ls_lookup-cr_bill_doc.
              IF sy-subrc IS INITIAL.
                COMMIT WORK AND WAIT.
              ENDIF.
            ENDIF.
            ls_update-cr_bill_doc = lv_bill1.
            MODIFY zsd_bag_dr_cr FROM ls_update.
            IF sy-subrc IS INITIAL.
              COMMIT WORK AND WAIT.
            ENDIF.
            MODIFY it_final FROM ls_final_temp1
                            TRANSPORTING crn_doc_no crn_doc_date credit_qty
                            WHERE vbeln = ls_final_temp1-vbeln
                              AND vgbel = ls_final_temp1-vgbel
                              AND sale_order = ls_final_temp1-sale_order.
            CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
              EXPORTING
                wait = 'X'.
            WAIT UP TO 1 SECONDS.
          ENDIF.
        WHEN 'DEBIT'.

          REFRESH: lt_datain, lt_return1, lt_success .
          CLEAR :  ls_datain-ship_to.
          ls_datain-doc_type   = 'ZSDR'.
          ls_datain-sold_to    = |{ ls_final_temp1-tdlnr ALPHA = IN }|.
          ls_datain-ref_item   = lv_int.
          ls_datain-origindoc  = ls_final_temp1-sale_order.
          ls_datain-item       = lv_int.
          APPEND ls_datain TO lt_datain.
          CLEAR ls_datain.

          REFRESH: lt_success, lt_return1.
          CALL FUNCTION 'BAPI_BILLINGDOC_CREATEMULTIPLE'
            EXPORTING
              creatordatain = ls_create
              testrun       = ' '
            TABLES
              billingdatain = lt_datain
              return        = lt_return1
              success       = lt_success.
          SORT lt_success BY ref_doc.
          CLEAR ls_success.
          READ TABLE lt_success INTO ls_success WITH KEY ref_doc = ls_final_temp1-sale_order BINARY SEARCH.
          IF sy-subrc = 0.
            DATA(lv_bill)      = ls_success-bill_doc.

            ls_final_temp1-drn_doc_date = sy-datum.
            ls_final_temp1-drn_doc_no = lv_bill.
            ls_final_temp1-debit_qty = ls_final_temp1-short_net_weight.

            ls_update-debit_note = ls_final_temp1-sale_order.
            CLEAR ls_lookup.
            READ TABLE lt_lookup_cr INTO ls_lookup WITH KEY debit_note = ls_update-debit_note BINARY SEARCH.
            IF sy-subrc IS INITIAL.
              UPDATE zsd_bag_dr_cr
              SET
               new_cr_dr = lv_bill
              WHERE vbeln = ls_final_temp1-vbeln
               AND vgbel = ls_final_temp1-vgbel
               AND debit_note = ls_final_temp1-sale_order
               AND db_bill_doc = ls_lookup-db_bill_doc.
              IF sy-subrc IS INITIAL.
                COMMIT WORK AND WAIT.
              ENDIF.
            ENDIF.
            ls_update-db_bill_doc = lv_bill.
            MODIFY zsd_bag_dr_cr FROM ls_update.
            IF sy-subrc IS INITIAL.
              COMMIT WORK AND WAIT.
            ENDIF.
            MODIFY it_final FROM ls_final_temp1
                            TRANSPORTING drn_doc_no drn_doc_date debit_qty
                            WHERE vbeln = ls_final_temp1-vbeln
                              AND vgbel = ls_final_temp1-vgbel
                              AND sale_order = ls_final_temp1-sale_order.
            CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
              EXPORTING
                wait = 'X'.
            WAIT UP TO 1 SECONDS.
          ENDIF.
      ENDCASE.
      CLEAR ls_final.
    ENDLOOP.

    CALL METHOD lo_grid->refresh_table_display( ).
    IF it_final IS NOT INITIAL.
      SELECT vbeln,
             netwr,
             mwsbp
             FROM vbrp
             INTO TABLE @DATA(it_credit_rate)
             FOR ALL ENTRIES IN @it_final
             WHERE vbeln = @it_final-crn_doc_no.
      IF sy-subrc = 0.
        SORT it_credit_rate BY vbeln.
      ENDIF.
      LOOP AT it_final ASSIGNING FIELD-SYMBOL(<fs_final>).
        READ TABLE it_credit_rate INTO DATA(ls_credit_rate) WITH KEY vbeln = <fs_final>-crn_doc_no BINARY SEARCH.
        IF sy-subrc = 0.
          <fs_final>-amt_crn_to_cust = ls_credit_rate-netwr + ls_credit_rate-mwsbp.
          UPDATE zsd_bag_dr_cr SET amt_crn_to_cust = <fs_final>-amt_crn_to_cust
           WHERE vbeln = <fs_final>-vbeln
             AND vgbel = <fs_final>-vgbel
             AND cr_bill_doc = <fs_final>-crn_doc_no.
          IF sy-subrc IS INITIAL.
            COMMIT WORK AND WAIT .
          ENDIF.
        ENDIF.
      ENDLOOP.
      IF <fs_final> IS ASSIGNED.
        UNASSIGN <fs_final>.
      ENDIF.
    ENDIF.
    CALL METHOD lo_grid->refresh_table_display( ).
  ENDMETHOD.

    ls_item-entry_uom  = ls_sto_temp3-sales_unit.
    ls_item-plant = rec_plant.
    ls_item-stge_loc = lv_storge_loc.
    ls_item-batch = lv_batch1.
    ls_item-stck_type = '3'.
    ls_item-ref_doc = ls_sto_temp3-vbeln.
    ls_item-gl_account = gl.
    ls_item-costcenter = cc.
    APPEND ls_item TO lt_items.
    CLEAR: ls_item, ls_sto_temp3.

  ENDLOOP.
  CALL FUNCTION 'BAPI_GOODSMVT_CREATE'
    EXPORTING
      goodsmvt_header  = ls_header
      goodsmvt_code    = ls_code
    IMPORTING
      goodsmvt_headret = ls_headret
      materialdocument = lv_mat_doc
      matdocumentyear  = lv_doc_year
    TABLES
      goodsmvt_item    = lt_items
      return           = lt_return.

  IF lv_mat_doc IS NOT INITIAL.
    CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'.
    WAIT UP TO 2 SECONDS.

    LOOP AT it_sto_temp INTO DATA(ls_sto_temp4) WHERE transp_code = ls_sto_temp-transp_code AND receiving_plant = ls_sto_temp-receiving_plant.
      MOVE-CORRESPONDING ls_sto_temp4 TO ls_update_sto.
      ls_update_sto-sales_doc = lv_mat_doc.
      ls_update_sto-idntification = 'DR_TRANSP'.

      MODIFY zsd_stk_transfer FROM ls_update_sto.
      IF sy-subrc IS INITIAL.
        COMMIT WORK AND WAIT.
      ENDIF.

      ls_sto_temp1-trnsporter_sales_odr = lv_mat_doc.
      ls_sto_temp1-idntification = 'DR_TRANSP'.
      ls_sto_temp1-trnsporter_qty = lv_max_qty.
      APPEND ls_sto_temp1 TO it_sto_final.

      ls_error-belnr = ls_sto_temp4-belnr.
      ls_error-msg = 'MIGO Done for writeback quantity'.
      ls_error-text = 'MIGO Done'.
      ls_error-type = 'Success'.
      ls_error-vgbel = ls_sto_temp4-vbeln.
      ls_error-del_migo_no = ls_sto_temp4-del_migo_no.
      APPEND ls_error TO it_error.
      CLEAR: ls_error, ls_sto_temp4.
    ENDLOOP.
  ELSE.
    CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
    LOOP AT it_sto_temp INTO DATA(ls_sto_temp5) WHERE transp_code = ls_sto_temp-transp_code AND receiving_plant = ls_sto_temp-receiving_plant.
      CLEAR: ls_header, ls_code, lt_items, lt_return.
      ls_error-belnr = ls_sto_temp5-belnr.
      ls_error-vgbel = ls_sto_temp5-vbeln.
      ls_error-del_migo_no = ls_sto_temp5-del_migo_no.
      ls_error-msg = 'MIGO Not Done for writeback quantity'.
      ls_error-text = 'MIGO Not Done for writeback quantity'.
      ls_error-type = 'Error'.
      APPEND ls_error TO it_error.
      CLEAR:ls_error, ls_sto_temp5.
    ENDLOOP.
  ENDIF.

  CLEAR lv_bal_qty1.
  lv_srno = lv_srno + 1.

ENDFORM.
