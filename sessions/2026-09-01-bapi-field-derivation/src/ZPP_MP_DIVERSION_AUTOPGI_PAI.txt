MODULE user_command_9000 INPUT.
  CASE sy-ucomm.
    WHEN 'FIG'.
      IF r_open_task IS NOT INITIAL.
        CLEAR gs_change.
*        IF sy-uname = 'IBMABAP30'.
        TYPES : BEGIN OF ty_t1,
                  checkbox           TYPE char1,
                  zdiversion_no      TYPE zdiversion_no,
                  zcreatedate        TYPE zcreatedate,
                  zcreatedon         TYPE zcreatedon,
                  parent_process_ord TYPE aufnr,
                  return_quantity    TYPE kwmeng,
                  zztm_reg           TYPE zde_tm_reg,
                  vbeln              TYPE vbeln_vl,
                  bill               TYPE vbeln_vf,
                END OF ty_t1.
        DATA: lt_zppt_div TYPE TABLE OF ty_t1,
              ls_zppt_div TYPE ty_t1.

        CLEAR lt_zppt_div.

        DATA lv_date1 TYPE sy-datum.
        lv_date1 = sy-datum.
        SUBTRACT 1 FROM lv_date1.

        SELECT a~zrmc_pltcd
                   FROM zta_rmc_plant AS a
                   WHERE zvalfrom LE @sy-datum
                   AND zvalto GE @sy-datum
                   INTO TABLE @lt_rmc_plant.
        IF sy-subrc = 0.
          SORT lt_rmc_plant BY zrmc_pltcd.
          DELETE ADJACENT DUPLICATES FROM lt_rmc_plant COMPARING zrmc_pltcd.
          CLEAR  r_dtm.
          LOOP AT lt_rmc_plant INTO ls_rmc_plant1.
            AUTHORITY-CHECK OBJECT 'ZRMC_COR1' ID 'WERKS' FIELD ls_rmc_plant1-zrmc_pltcd.
            IF sy-subrc = 0.
              APPEND VALUE #( sign = 'I' option = 'EQ' low = ls_rmc_plant1-zrmc_pltcd  ) TO r_dtm.
            ENDIF.
          ENDLOOP.
        ENDIF.
        IF r_dtm IS NOT INITIAL.
          SELECT a~zdiversion_no,
                 a~zcreatedate,
                 a~zcreatedon,
                 a~parent_process_ord,
                 a~return_quantity,
                 a~order_number
              FROM zppt_diversion AS a
              INNER JOIN caufv AS caufv ON caufv~aufnr =  a~parent_process_ord
              INTO TABLE @DATA(lt_zppt_diversion1)
              WHERE order_number = ' '
              AND caufv~werks IN @r_dtm
              AND  zdiversion_no NE ' '
              AND zcreatedate NE '00000000'
               AND zcreatedate GE @lv_date1 AND  zcreatedate LE @sy-datum.
          IF sy-subrc = 0.
            SELECT a~zdiversion_no,
                 a~zcreatedate,
                 a~zcreatedon,

                 a~return_quantity,

             INNER JOIN caufv AS caufv ON caufv~aufnr =  a~parent_process_ord

                AND zcreatedate NE '00000000'
               AND zcreatedate GE @lv_date1 AND  zcreatedate LE @sy-datum

          ENDIF.

              SELECT SINGLE *
                      FROM zrmc_so_details
                      INTO @DATA(ls_zrmc_so_details)
                      WHERE aufnr = @ls_zppt_diversion1-order_number

                       AND bill_no NE @space.

              ENDIF.

              CLEAR ls_zppt_div.
            ENDLOOP.

                                  b~matnr , b~fkimg ,b~werks
                                  FROM vbfa AS a
                                  INNER JOIN vbrp AS b ON b~vbeln EQ a~vbeln

                                  WHERE a~vbelv = @ls_likp-vbeln
                                    AND a~vbtyp_n = 'M'.
                            ENDIF.
                            IF sy-subrc = 0.
                              SELECT SINGLE a~*
                                     FROM vbrp AS a
                               INNER JOIN vbrk AS b ON a~vbeln = b~vbeln
                              INTO @ls_vbrp_m
                              WHERE a~vbeln = @ls_vbfa-vbeln
                               AND vbtyp_ana = 'U'
                               AND rfbsk NE 'E'.
                              IF sy-subrc NE 0.
                                SELECT SINGLE a~*
                                                    FROM vbrp AS a
                                              INNER JOIN vbrk AS b ON a~vbeln = b~vbeln
                                             INTO @ls_vbrp_m
                                             WHERE a~vbeln = @ls_vbfa-vbeln
                                              AND vbtyp_ana = 'M'
                                              AND fksto NE 'X'.
                              ENDIF.
                              IF sy-subrc = 0.
                                gv_vbeln = ls_vbfa-vbeln.
                                gv_matnr = ls_vbfa-matnr.
                                gv_werks = ls_vbfa-werks.
*                gs_change-material = ls_vbfa-matnr.
*                gs_change-plant = ls_vbfa-werks.
                                IF gs_change-new_sale_order IS NOT INITIAL.
                                  lv_sale_new = |{ gs_change-new_sale_order ALPHA = IN }|.
                                  SELECT SINGLE matnr , werks FROM vbap INTO @ls_new_sale_order
                                    WHERE vbeln = @lv_sale_new AND posnr = '000010'.
                                  IF sy-subrc = 0.
                                    gs_change-material = ls_new_sale_order-matnr.
                                    gs_change-plant = ls_new_sale_order-werks.
                                  ENDIF.
                                ENDIF.
                                IF gs_change-sales_order IS NOT INITIAL.
                                  lv_sale_new = |{ gs_change-sales_order  ALPHA = IN }|.
                                  SELECT SINGLE matnr , werks FROM vbap INTO @ls_new_sale_order
                                    WHERE vbeln = @lv_sale_new AND posnr = '000010'.
                                  IF sy-subrc = 0.
                                    gs_change-material = ls_new_sale_order-matnr.
                                    gs_change-plant = ls_new_sale_order-werks.
                                  ENDIF.
                                ENDIF.
                                IF gs_change-return_quantity GT ls_vbfa-fkimg.
                                  MESSAGE 'Invoiced QTY and Return Quantity not matching'  TYPE 'E' DISPLAY LIKE 'E'.
                                  EXIT.
                                ELSE.
                                  gv_dis = 'X'.
                                ENDIF.
                              ELSE.
                                MESSAGE 'Invoice is not valid as Document Category not M'  TYPE 'E' DISPLAY LIKE 'E'.
                                EXIT.
                              ENDIF.
                            ELSE.
                              MESSAGE 'Invoice/Delivery Challan is not created yet'  TYPE 'E' DISPLAY LIKE 'E'.
                              EXIT.
                            ENDIF.
                          ELSE.
                            MESSAGE 'This Order is not linked with Delivery'  TYPE 'E' DISPLAY LIKE 'E'.
                            EXIT.
                          ENDIF.
                        ELSE.
*          MESSAGE 'Quantity is more than procees order QTY'  TYPE 'E' DISPLAY LIKE 'E'.
                          MESSAGE 'Return Quantity should be more than ZERO'  TYPE 'E' DISPLAY LIKE 'E'.
                          EXIT.
                        ENDIF.
                      ELSE.
*        MESSAGE 'This Order is not Valid' TYPE 'E'.
                        MESSAGE 'This Order is not Valid' TYPE 'E' DISPLAY LIKE 'E'.
                        EXIT.
                      ENDIF.
                    ENDIF.
                    IF gv_dis = 'X' AND gs_change-new_sale_order IS NOT INITIAL
                    AND gs_change-to_be_pro_ord_qty IS NOT INITIAL.
                      gv_pop = 'X'.
                    ENDIF.

                    IF gs_change-full_div_ind = 'X'.
                      IF gs_change-full_div_qty IS INITIAL.
                        MESSAGE 'If Full diversion indicator is ticked then Full Diversion QTY is mandatory to fill' TYPE 'E' DISPLAY LIKE 'E'.
                        EXIT.
                      ELSE.
                        IF gs_change-full_div_qty GT gs_change-to_be_pro_ord_qty.
                          MESSAGE 'Full Diversion QTY can not be greater than to be process QTY' TYPE 'E' DISPLAY LIKE 'E'.
                          EXIT.
                        ELSE.
                          lv_flag_fdi = 'X'.
                        ENDIF.
                      ENDIF.
                    ELSE.
                      lv_flag_fdi = 'X'.
                    ENDIF.

                    CLEAR gv_process.
                    IF gs_change-order_number IS NOT INITIAL.

                             b~meins AS quantity_uom,
                             b~kdauf AS sales_order,

                             c~gltrp AS basic_end_date,
                             d~zztm_code,
                             d~zztm_reg,
                             d~zztm_driver,

                             d~zztm_cap,
                             d~zztm_uom,

                             d~zzbatch_size,

                             d~zzmix_time,

                             d~zzmax_tm_size,
                             d~zzapproved,

      CALL FUNCTION 'BAPI_TRANSACTION_ROLLBACK'.
      ls_logs-field = ''.
      ls_logs-mesg = ls_ret-message.
      ls_logs-status = 'Error'.
      ls_logs-icon = '@0A@'. "Error
      APPEND ls_logs TO it_logs.
      CLEAR: ls_logs.
      CALL SCREEN '9001'.

    ELSE.
      gs_change-mat_doc = lv_doc.
      gs_change-doc_year = lv_yer.
      MODIFY zppt_diversion FROM gs_change.
      CALL FUNCTION 'BAPI_TRANSACTION_COMMIT'
        EXPORTING
          wait = abap_true.
      WAIT UP TO 1 SECONDS.
    ENDIF.

    gs_change-mat_doc = lv_doc.
    gs_change-doc_year = lv_yer.

*&---------------------------------------------------------------------*
*&      Module  ZFULL_DIV_QTY  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE zfull_div_qty INPUT.
  IF sy-ucomm = 'BACK'
   OR sy-ucomm = 'CANCEL'
   OR sy-ucomm = 'EXIT'
   OR sy-ucomm = '&REFRESH'.
  ELSE.
    IF sy-ucomm = ' '.
      IF gs_change-full_div_ind IS NOT INITIAL.
        IF gs_change-full_div_qty IS INITIAL.
          MESSAGE 'Full Diversion is mandatory' TYPE 'E'  DISPLAY LIKE 'E'.
          EXIT.
        ENDIF.
      ENDIF.
      IF gs_change-full_div_qty IS NOT INITIAL.
        IF gs_change-full_div_ind IS INITIAL.
          MESSAGE 'Full Diversion Tick is mandatory' TYPE 'E'  DISPLAY LIKE 'E'.
          EXIT.
        ENDIF.
      ENDIF.
*    ELSEIF sy-ucomm = 'RCCO'.
      IF gs_change-full_div_ind = 'X'.
        IF gs_change-full_div_qty IS INITIAL.
          MESSAGE 'If Full diversion indicator is ticked then Full Diversion QTY is mandatory to fill' TYPE 'E' DISPLAY LIKE 'E'.
          EXIT.
        ELSE.
          IF gs_change-full_div_qty GT gs_change-to_be_pro_ord_qty.
            MESSAGE 'Full Diversion QTY can not be greater than to be process QTY' TYPE 'E' DISPLAY LIKE 'E'.
            EXIT.
          ELSE.
            lv_flag_fdi = 'X'.
          ENDIF.
        ENDIF.
      ELSE.
        lv_flag_fdi = 'X'.
      ENDIF.
    ENDIF.
  ENDIF.
ENDMODULE.
*&---------------------------------------------------------------------*
*&      Module  ZQUANTITY  INPUT
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
MODULE zquantity INPUT.
  IF sy-ucomm = '' OR sy-ucomm = 'PNO'.
    IF gs_change-quantity GT 0 AND gs_change-zzreturn_qty GT 0.
      IF gs_change-quantity LE gs_change-zzreturn_qty.
        gs_change-full_div_ind = 'X'.
        gs_change-full_div_qty = gs_change-zzreturn_qty.
      ELSE.
        CLEAR:gs_change-full_div_ind,
              gs_change-full_div_qty .
      ENDIF.
    ENDIF.
  ENDIF.

  IF sy-ucomm = 'BACK'
   OR sy-ucomm = 'CANCEL'
   OR sy-ucomm = 'EXIT'
   OR sy-ucomm = '&REFRESH'.
  ELSE.

    IF gs_change-quantity IS NOT INITIAL AND gs_change-zzreturn_qty IS NOT INITIAL.
      IF gs_change-quantity LT gs_change-zzreturn_qty.
        MESSAGE 'Process Order Quantity can not be less than Return QTY'  TYPE 'E' DISPLAY LIKE 'E'.
        EXIT.
      ENDIF.
    ENDIF.
  ENDIF.

ENDMODULE.
