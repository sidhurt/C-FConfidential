*  -->  P1        TEXT
*  <--  P2        TEXT

 FORM write_bapi_error_messages .
   LOOP AT it_return ASSIGNING <fs_return>.

     APPEND INITIAL LINE TO it_message ASSIGNING <fs_message>.
     <fs_message>-matnr = <fs_header>-material.
     <fs_message>-werks = <fs_header>-plant.

     IF ( <fs_return>-type EQ 'E' OR <fs_return>-type EQ 'A' ).
       <fs_message>-light = '@0A@'.
     ELSE.
       <fs_message>-light = '@09@'.
     ENDIF.
     <fs_message>-index = gv_index.
     IF <fs_return>-message IS NOT INITIAL.
       <fs_message>-message = <fs_return>-message.
     ELSE.
       SELECT SINGLE text FROM t100 INTO <fs_message>-message
         WHERE sprsl = sy-langu
         AND   arbgb = <fs_return>-id
         AND   msgnr = <fs_return>-number.
     ENDIF.
   ENDLOOP.

 ENDFORM. " WRITE_BAPI_ERROR_MESSAGES

   READ TABLE it_header ASSIGNING <fs_header> INDEX 1.
   IF sy-subrc = 0 AND <fs_header>-plant IS NOT INITIAL.

     " Check for Depot Plant
     READ TABLE it_1iexgrps ASSIGNING <fs_1iexgrps>
            WITH KEY j_1iwerks = <fs_header>-plant.
     IF sy-subrc = 0 AND <fs_1iexgrps>-j_1idepot IS INITIAL.

       " Exclude POST Function Code in case of non depot plants.
       SET PF-STATUS 'STANDARD' EXCLUDING 'POST' IMMEDIATELY .

     ELSE.

       SET PF-STATUS 'STANDARD'.

     ENDIF.

   ELSE.

     " Exclude POST Function Code in case of non depot plants.
     SET PF-STATUS 'STANDARD'   EXCLUDING 'POST' IMMEDIATELY .

   ENDIF.

 ENDFORM. "PF_STATUS
