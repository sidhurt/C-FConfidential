FORM f_ucomm USING lv_ucomm LIKE sy-ucomm
                                  lv_selfield TYPE slis_selfield.


  IF gcl_ref IS INITIAL.
    CALL FUNCTION 'GET_GLOBALS_FROM_SLVC_FULLSCR'
      IMPORTING
        e_grid = gcl_ref.
  ENDIF.

  IF NOT gcl_ref IS INITIAL.
    CALL METHOD gcl_ref->check_changed_data.
  ENDIF.

  CASE lv_ucomm.
    WHEN gc_all."'/&ALL'.
      LOOP AT gt_rb1_final INTO gw_rb1_final.
        gw_rb1_final-zchk = abap_true.
        MODIFY gt_rb1_final FROM gw_rb1_final.
      ENDLOOP.

      lv_selfield-refresh = abap_true.

    WHEN gc_calc.

      LOOP AT gt_rb1_final INTO gw_rb1_final.
        gw_rb1_final-final_grq = gw_rb1_final-receivedqty + gw_rb1_final-net_wt.
        MODIFY gt_rb1_final FROM gw_rb1_final.
      ENDLOOP.

      lv_selfield-refresh = abap_true.

    WHEN gc_sal."'/&SAL'.
      LOOP AT gt_rb1_final INTO gw_rb1_final.
        gw_rb1_final-zchk = gc_space."' '.
        MODIFY gt_rb1_final FROM gw_rb1_final.
      ENDLOOP.

      lv_selfield-refresh = abap_true.

    WHEN gc_post.
*
      PERFORM f_popup.
      IF lv_yes = abap_true.
        PERFORM f_bapi.
      ENDIF.
      lv_selfield-refresh = abap_true.

    WHEN gc_post1.
      PERFORM f_popup.
      IF lv_yes = abap_true.
        PERFORM f_collective.
      ENDIF.
      lv_selfield-refresh = abap_true.

    WHEN gc_gr.
      PERFORM f_popup.
      IF lv_yes = abap_true.
        PERFORM gr_at_plant.
      ENDIF.
      lv_selfield-refresh = abap_true.

    WHEN gc_cgr.
      PERFORM f_popup.
      IF lv_yes = abap_true.
        PERFORM c_gr_at_plant.
      ENDIF.
      lv_selfield-refresh = abap_true.

    WHEN gc_disp."'DISP_LOG'.

      PERFORM f_displog.

    WHEN OTHERS.
      LEAVE TO SCREEN 0.
  ENDCASE.
ENDFORM.
