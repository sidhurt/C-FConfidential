  METHOD change_to_local_timezone.
    DATA: lv_system_timezone TYPE tznzone.

    CALL FUNCTION 'GET_SYSTEM_TIMEZONE'
      IMPORTING
        timezone            = lv_system_timezone
      EXCEPTIONS
        customizing_missing = 1
        OTHERS              = 2.
    IF sy-subrc <> 0.
      lv_system_timezone = 'UTC'.
    ENDIF.

    DATA(lo_tz_handler) = NEW cl_le_dlv_struc_time_zone_conv(
      it_system_loc_dt_fields    = VALUE #( ( date_field_name = 'CreationDate'               time_field_name = 'CreationTime' ) )
      it_shipping_loc_dt_fields  = VALUE #( ( date_field_name = 'PickingDate'                time_field_name = 'PickingTime' )
                                            ( date_field_name = 'PlannedGoodsIssueDate'      time_field_name = 'GoodsIssueTime' )
                                            ( date_field_name = 'TransportationPlanningDate' time_field_name = 'TransportationPlanningTime' )
                                            ( date_field_name = 'LoadingDate'                time_field_name = 'LoadingTime' )
                                            ( date_field_name = 'ActualGoodsMovementDate'    time_field_name = 'ActualGoodsMovementTime' )
                                            ( date_field_name = 'ProofOfDeliveryDate'        time_field_name = 'ConfirmationTime' ) )
      it_receiving_loc_dt_fields = VALUE #( ( date_field_name = 'DeliveryDate'               time_field_name = 'DeliveryTime ' ) ) ).

    lo_tz_handler->convert_utc_to_local(
      EXPORTING
        iv_system_loc_time_zone    = lv_system_timezone
        iv_shipping_loc_time_zone  = cs_entity-shippinglocationtimezone
        iv_receiving_loc_time_zone = cs_entity-receivinglocationtimezone
      IMPORTING
        et_messages = DATA(lt_conv_messages)
      CHANGING
        cs_structure = cs_entity ).

    READ TABLE lt_conv_messages INDEX 1 ASSIGNING FIELD-SYMBOL(<fs_conv_message>).
    IF sy-subrc = 0.
      DATA(lo_api_exception) = NEW cx_od_api_exception( ).
      MOVE-CORRESPONDING <fs_conv_message> TO lo_api_exception->system_fields.
      lo_api_exception->system_fields-subrc = 1.
      RAISE EXCEPTION lo_api_exception.
    ENDIF.
  ENDMETHOD.
