SOURCE_BEGIN|CL_API_OUTBOUND_DELIVE_DPC_EXT|/IWBEP/IF_MGW_APPL_SRV_RUNTIME~CREATE_DEEP_ENTITY
  METHOD /iwbep/if_mgw_appl_srv_runtime~create_deep_entity.
    DATA: BEGIN OF ls_serials.
            INCLUDE TYPE cl_api_outbound_delive_mpc=>ts_a_serialnmbrdeliverytype.
    DATA:   to_maintenanceitemobject TYPE STANDARD TABLE OF cl_api_outbound_delive_mpc=>ts_a_maintenanceitemobjecttype WITH DEFAULT KEY,
          END OF ls_serials.

    DATA: BEGIN OF ls_items.
            INCLUDE TYPE cl_api_outbound_delive_mpc=>ts_a_outbdeliveryitemtype.
    DATA:   to_serialdeliveryitem LIKE ls_serials,
          END OF ls_items.

    DATA: BEGIN OF ls_headers.
            INCLUDE TYPE cl_api_outbound_delive_mpc=>ts_a_outbdeliveryheadertype.
    DATA:   to_deliverydocumentitem LIKE TABLE OF ls_items,
          END OF ls_headers.

    DATA: lv_entityset_name TYPE /iwbep/mgw_tech_name.

    DATA: ls_sales_order_items TYPE shp_bapidlvreftosalesorder_ext,
          lt_sales_order_items TYPE TABLE OF shp_bapidlvreftosalesorder_ext.

    DATA: ls_serial_numbers TYPE bapidlvserialnumber,
          lt_serial_numbers TYPE TABLE OF bapidlvserialnumber.

    DATA: lt_created_items TYPE TABLE OF bapidlvitemcreated.

    DATA: lv_ship_point        TYPE bapidlvcreateheader-ship_point,
          ls_mobj              TYPE cl_api_outbound_delive_mpc=>ts_a_maintenanceitemobjecttype,
          lt_return            TYPE TABLE OF bapiret2,
          lv_number_deliveries TYPE bapidlvcreateheader-num_deliveries,
          lv_delivery          TYPE bapishpdelivnumb-deliv_numb,
          lo_exception         TYPE REF TO cx_od_api_exception.

    DATA: ls_lips     TYPE lipsvb,
          ls_likp     TYPE likpvb,
          lt_lips     TYPE shp_lips_t,
          lt_lips_all TYPE shp_lips_t,
          lt_likp     TYPE shp_likp_t.

    DATA: ls_likpvb             TYPE likpvb,
          ls_outbdeliveryheader TYPE a_outbdeliveryheader_1,
          lv_system_tz          TYPE timezone.

    lv_entityset_name = io_tech_request_context->get_entity_set_name( ).
    CASE lv_entityset_name.
      WHEN 'A_OutbDeliveryHeader'.
        io_data_provider->read_entry_data(
          IMPORTING
            es_data = ls_headers
        ).

        " Fill structure from request
        lv_ship_point = ls_headers-shippingpoint.
        LOOP AT ls_headers-to_deliverydocumentitem ASSIGNING FIELD-SYMBOL(<item>).
          ls_sales_order_items-deliv_numb = <item>-deliverydocument.
          ls_sales_order_items-dlv_qty    = <item>-actualdeliveryquantity.
          ls_sales_order_items-ref_doc    = <item>-referencesddocument.
          ls_sales_order_items-ref_item   = <item>-referencesddocumentitem.
          ls_sales_order_items-sales_unit = <item>-deliveryquantityunit.

          LOOP AT <item>-to_serialdeliveryitem-to_maintenanceitemobject INTO ls_mobj.
            ls_serial_numbers-ref_doc  = ls_sales_order_items-ref_doc.
            ls_serial_numbers-ref_item = ls_sales_order_items-ref_item.
            ls_serial_numbers-serialno = ls_mobj-serialnumber.
            APPEND ls_serial_numbers TO lt_serial_numbers.
          ENDLOOP.

          " Fill extension fields
          IF ls_headers-ext IS NOT INITIAL.
            MOVE-CORRESPONDING ls_headers-ext TO ls_sales_order_items.
          ENDIF.

          IF <item>-ext IS NOT INITIAL.
            MOVE-CORRESPONDING <item>-ext TO ls_sales_order_items.
            CLEAR <item>-ext.
          ENDIF.

          APPEND ls_sales_order_items TO lt_sales_order_items.
          CLEAR ls_sales_order_items.
        ENDLOOP.
        CLEAR ls_headers-ext.

        " Check if input data contains only allowed properties
        DATA itab TYPE TABLE OF string.
        cl_api_outbound_delive_mpc_ext=>get_properties_create(
          IMPORTING
            ex_whitelist = itab
        ).
        check_whitelist(
          EXPORTING
            im_struct    = ls_headers
            im_whitelist = itab
        ).

        TRY .
            DATA(lv_bapi_type) = check_document_type( im_sddocument = ls_headers-to_deliverydocumentitem[ 1 ]-referencesddocument ).
            CALL METHOD /spe/cl_inb_action_info=>set_api_info
              EXPORTING
                iv_calling_service = |API_OUTBOUND_DELIVERY_SRV|
                iv_var_1           = |J|.
            IF lv_bapi_type = 'SLS'.
              DATA(lo_od_iref) = cl_outbound_delivery_factory=>get_instance( ).
              lo_od_iref->cd_with_ref_specific_item_s(
                EXPORTING
                  im_ship_point        = ls_headers-shippingpoint    " Shipping Point/Receiving Point
                IMPORTING
                  ex_delivery          = lv_delivery    " Delivery
                  ex_num_deliveries    = lv_number_deliveries    " Number of created documents
                  ex_return            = lt_return
                CHANGING
                  ch_sales_order_items = lt_sales_order_items
                  ch_serial_numbers    = lt_serial_numbers
                  ch_created_items     = lt_created_items
              ).

            ELSEIF lv_bapi_type = 'STO'.
              DATA lt_sto_items TYPE TABLE OF shp_bapidlvreftosto_ext.
              MOVE-CORRESPONDING lt_sales_order_items TO lt_sto_items.

              lo_od_iref = cl_outbound_delivery_factory=>get_instance( ).
              lo_od_iref->cd_sto_specific_items(
                EXPORTING
                  im_ship_point     = ls_headers-shippingpoint
                IMPORTING
                  ex_delivery       = lv_delivery
                  ex_num_deliveries = lv_number_deliveries
                  ex_return         = lt_return
                CHANGING
                  ch_sto_items      = lt_sto_items
                  ch_serial_numbers = lt_serial_numbers
                  ch_created_items  = lt_created_items
              ).
            ENDIF.
          CATCH cx_od_api_exception INTO lo_exception.
            IF lo_exception->system_fields-subrc EQ 1.
              mo_context->get_message_container( )->add_message(
                          EXPORTING
                            iv_msg_type               = lo_exception->system_fields-msgty    " Message Type
                            iv_msg_id                 = lo_exception->system_fields-msgid    " Message Class
                            iv_msg_number             = lo_exception->system_fields-msgno    " Message Number
                            iv_msg_v1                 = lo_exception->system_fields-msgv1    " Message Variable
                            iv_msg_v2                 = lo_exception->system_fields-msgv2    " Message Variable
                            iv_msg_v3                 = lo_exception->system_fields-msgv3    " Message Variable
                            iv_msg_v4                 = lo_exception->system_fields-msgv4    " Message Variable
              ).
              RAISE EXCEPTION TYPE /iwbep/cx_mgw_busi_exception
                EXPORTING
                  textid            = /iwbep/cx_mgw_busi_exception=>business_error
                  message_container = mo_context->get_message_container( ).
            ELSE.
              LOOP AT lo_exception->return_bapi TRANSPORTING NO FIELDS
                              WHERE type CA 'AEX'.
                EXIT.
              ENDLOOP.

              IF sy-subrc EQ 0.
                mo_context->get_message_container( )->add_messages_from_bapi(
                    iv_determine_leading_msg  = /iwbep/if_message_container=>gcs_leading_msg_search_option-first
                    it_bapi_messages          = lo_exception->return_bapi    " Return parameter table

                ).

                RAISE EXCEPTION TYPE /iwbep/cx_mgw_busi_exception
                  EXPORTING
                    textid            = /iwbep/cx_mgw_busi_exception=>business_error
                    message_container = mo_context->get_message_container( ).
              ENDIF.
            ENDIF.
        ENDTRY.

        " Fill return structure of deep create operation.
        CALL FUNCTION 'LE_DELIVERY_READ'
          EXPORTING
            iv_vbeln              = lv_delivery
            nicht_sperren_1       = 'X'
            iv_get_deliv_buffered = 'X'
            iv_no_init            = 'X'
            iv_no_read            = 'X'
            iv_no_buffer_refresh  = 'X'
          IMPORTING
            es_likpvb             = ls_likpvb.

        TRY.
            cl_le_shp_gfn_mapper=>map_delivery_header_to_gfn(
              EXPORTING
                is_delivery_header     = ls_likpvb
              CHANGING
                cs_delivery_header_gfn = ls_outbdeliveryheader
            ).
            DATA(lo_tz_handler) = NEW cl_le_dlv_struc_time_zone_conv(
              it_system_loc_dt_fields    = VALUE #( ( date_field_name = 'CreationDate' time_field_name = 'CreationTime' ) )
              it_shipping_loc_dt_fields  = VALUE #( ( date_field_name = 'PickingDate'                 time_field_name = 'PickingTime' )
                                                    ( date_field_name = 'PlannedGoodsIssueDate'       time_field_name = 'GoodsIssueTime' )
                                                    ( date_field_name = 'TransportationPlanningDate'  time_field_name = 'TransportationPlanningTime' )
                                                    ( date_field_name = 'LoadingDate'                 time_field_name = 'LoadingTime' ) )
              it_receiving_loc_dt_fields = VALUE #( ( date_field_name = 'DeliveryDate' time_field_name = 'DeliveryTime ' ) )
            ).
            CALL FUNCTION 'GET_SYSTEM_TIMEZONE'
              IMPORTING
                timezone = lv_system_tz.
            lo_tz_handler->convert_to_utc(
              EXPORTING
                iv_system_loc_time_zone    = lv_system_tz
                iv_shipping_loc_time_zone  = ls_outbdeliveryheader-shippinglocationtimezone
                iv_receiving_loc_time_zone = ls_outbdeliveryheader-receivinglocationtimezone
              CHANGING
                cs_structure               = ls_outbdeliveryheader
            ).
          CATCH cx_corr_dyn_error.
            " Fallback for safety reasons
            SELECT
              SINGLE *
            FROM
              a_outbdeliveryheader_1
            WHERE
              a_outbdeliveryheader_1~deliverydocument = @lv_delivery
            INTO @ls_outbdeliveryheader.
        ENDTRY.

        copy_data_to_ref(
          EXPORTING
            is_data = ls_outbdeliveryheader
          CHANGING
            cr_data = er_deep_entity
        ).
    ENDCASE.
  ENDMETHOD.
