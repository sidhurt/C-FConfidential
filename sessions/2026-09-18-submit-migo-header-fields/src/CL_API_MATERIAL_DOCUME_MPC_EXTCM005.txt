    "Looping the table for creating Function Imports documentations
    LOOP AT it_function_anno INTO DATA(ls_function_anno).
      "Please just change the Schema Namespace and Entity Container Name as per your API in the code line below.
      DATA(lv_target) = 'API_MATERIAL_DOCUMENT_SRV.' && 'API_MATERIAL_DOCUMENT_SRV_Entities/'
